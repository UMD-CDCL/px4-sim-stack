# nvinfer: the primary detector.
#
# The default is the ResNet18 TrafficCamNet model that ships with DeepStream.
# It detects car, bicycle, person and road_sign. "person" is the class that
# matters for this stack, and the model needs no download, so the stack detects
# something the first time you start it.
#
# It is a starting point, not the right model for casualty detection. A person
# who is prone is out of distribution for TrafficCamNet, which was trained on
# upright pedestrians seen from a traffic camera. See modules/perception/models/
# for how to put your own model here.
#
# Values below come from the DeepStream sample config, with two changes:
#   batch-size  30 -> 1     one camera, not thirty
#   engine file          -> a path in the cache volume, so it survives restarts

[property]
gpu-id=0
net-scale-factor=0.00392156862745098
onnx-file=/opt/perception/models/Primary_Detector/resnet18_trafficcamnet_pruned.onnx
# nvinfer generates this name itself and saves next to the ONNX. Naming the
# same path here lets the next start reuse it instead of rebuilding.
model-engine-file=/opt/perception/models/Primary_Detector/resnet18_trafficcamnet_pruned.onnx_b2_gpu0_fp16.engine
labelfile-path=/opt/perception/models/Primary_Detector/labels.txt
int8-calib-file=/opt/perception/models/Primary_Detector/cal_trt.bin
# Must match the streammux batch size, which is one slot per camera.
batch-size=2
process-mode=1
model-color-format=0
# 0=FP32 1=INT8 2=FP16. The RTX 3070 has fast FP16 and the accuracy loss is
# smaller than INT8.
network-mode=2
num-detected-classes=4
interval=0
gie-unique-id=1
cluster-mode=2

# Report only people. Remove this line to see cars and road signs as well.
# Class order comes from labels.txt: 0 car, 1 bicycle, 2 person, 3 road_sign.
filter-out-class-ids=0;1;3

[class-attrs-all]
topk=20
nms-iou-threshold=0.5
pre-cluster-threshold=0.2

# Person. A lower threshold finds more and reports more false positives. Raise
# it once you know what your scene looks like.
[class-attrs-2]
topk=20
nms-iou-threshold=0.5
pre-cluster-threshold=0.3
