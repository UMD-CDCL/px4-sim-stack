# nvmsgconv: what goes into the message besides the boxes.
#
# The minimal payload schema, which this stack uses, carries the sensor id and
# the object list. The place and analytics blocks below are only read by the
# full schema (msg-conv-payload-type=0). They stay here so that switching to
# the full schema needs no new file.

# The id becomes "sensorId" in the payload, which is how the ROS side tells the
# cameras apart.
#
# One sensor, because there is one pipeline for each camera and each pipeline
# has one source. The entrypoint expands DS_SENSOR_ID differently for each, so
# the two processes read the same file and produce different ids.
[sensor0]
enable=1
type=Camera
id=${DS_SENSOR_ID}
location=0;0;0
description=${DS_SENSOR_ID} camera on the recon airframe
coordinate=0;0;0

[place0]
enable=1
id=0
type=field
name=recon_field
location=0;0;0
coordinate=0;0;0

[analytics0]
enable=1
id=px4simstack
description=DeepStream primary detector
source=px4-sim-stack
version=1.0
